# Pipeline Activity Panel — Implementation Plan (v2)

## Goal

Build a collapsible Pipeline Activity view in the right sidebar that acts as an engineering execution trace. Answers: What phase is CircuitBot working on? Is it progressing, waiting, retrying, or failing? How long has it taken? What result did the stage produce?

**Two-phase delivery:** Phase 1 = authoritative tracking/event/reconnect layer. Phase 2 = panel UI.

---

## Phase 1: Backend Tracking & Event Layer

### 1A. New file: `agent/pipeline_tracker.py` (~180 lines)

**Domain model:**

```python
@dataclass
class StageDefinition:
    key: str              # node_name in LangGraph (e.g. "constraint_checker")
    label: str            # display name (e.g. "Constraint Check")
    phase: str            # phase group key (see below)
    order: int            # canonical display order (1-34)
    optional: bool = False  # may be skipped (e.g. clarify, ask_pcb_approval)
    graph_versions: tuple = ("new",)  # which graphs include this stage

@dataclass
class StageAttempt:
    attempt: int
    status: str           # pending|running|waiting|completed|warning|failed|skipped|cancelled
    started_at: float     # monotonic
    completed_at: float | None
    wall_started: float   # time.time() for transport
    wall_completed: float | None
    summary: str
    metrics: dict         # structured, whitelisted fields only

@dataclass
class PipelineRun:
    run_id: str
    graph_version: str    # "new" | "legacy"
    status: str           # pending|running|completed|failed|cancelled
    stages: dict[str, list[StageAttempt]]  # stage_key → attempts list
    current_stage: str | None
    sequence: int         # monotonic event counter
    wall_started: float
    wall_completed: float | None
```

**Phase groups** (professional labels, not node-level granularity):

| Phase | Label | Stages |
|-------|-------|--------|
| `understand` | Understand | clarify, analyze, research, architecture_planner, capability_resolver |
| `components` | Components | select, dependency_expander, deduplicator, constraint_checker, repair, validate, ask_validation_help, post_validate_constraint_checker, freeze_components, datasheet_search |
| `schematic` | Schematic | symbol_compatibility, dispatch, pin_marker, symbol_validate, connection_search, netlist, power_net_repair, structural_net_validate, structural_net_repair, placement, routing, connectivity_validate, connectivity_repair, schematic_audit, schematic_repair |
| `pcb` | PCB | ask_pcb_approval, ask_board_config, pcb_layout, design_review |

**Key functions:**

- `track_node(node_key, node_fn)` — returns a wrapped function that:
  - Calls `emit(config, "pipeline", {...})` with actions: `stage_started`, `stage_updated`, `stage_finished`
  - Allocates incrementing attempt numbers per stage_key
  - Uses `time.monotonic()` for durations, `time.time()` for transport timestamps
  - Preserves original return value exactly
  - On exception: emits `stage_finished` with status=failed, re-raises with original traceback
  - Approval nodes (`ask_pcb_approval`, `ask_board_config`) transition running→waiting→completed|skipped
  - Never serializes complete LangGraph state — only whitelisted summaries

- `get_current_run()` — returns the active `PipelineRun` for the current thread (stored in `threading.local()`)
- `init_run(graph_version)` — creates a new `PipelineRun`, stores in thread-local
- `close_run(status)` — marks run terminal, emits `run_finished`

**Stage summary extraction** (whitelisted, per node):

```python
STAGE_SUMMARIES = {
```

### 2B. New file: `static/pipeline-panel.js` (~300 lines)

Reducer-driven module:

```javascript
(function() {
  'use strict';
  
  // ── State ──
  let state = {
    runId: null,
    graphVersion: null,
    runStatus: 'idle',     // idle|running|completed|failed|cancelled
    phases: {},            // phase_key → {label, stages: {key → {attempts[], currentAttempt, status, label}}}
    activePhase: null,
    expandedStages: new Set(),
    sequence: 0,           // last processed sequence number
    wallStarted: null,
    events: [],            // capped at 500 for correlated log display
  };
  
  // ── Reducer ──
  function reducePipelineEvent(state, event) {
    if (event.run_id !== state.runId && state.runId) return state; // stale run
    if (event.sequence <= state.sequence && state.sequence > 0) return state; // duplicate
    
    switch (event.action) {
      case 'run_started':
        return { ...initialState(), runId: event.run_id, graphVersion: event.graph_version, runStatus: 'running', wallStarted: event.wall_started || Date.now()/1000 };
      case 'stage_started':
        return handleStageStarted(state, event);
      case 'stage_finished':
        return handleStageFinished(state, event);
      case 'run_finished':
        return { ...state, runStatus: event.status, sequence: event.sequence };
      default:
        return state;
    }
  }
  
  // ── Socket integration ──
  function attachSocket(socket) {
    socket.on('pipeline', (data) => {
      state = reducePipelineEvent(state, data);
      render();
    });
    socket.on('agent:log', (data) => {
      if (data.stage_key && data.attempt) {
        appendCorrelatedLog(data.stage_key, data.attempt, data.message);
      }
    });
    socket.on('agent:done', () => { /* update run status */ });
    socket.on('agent:error', () => { /* update run status */ });
  }
  
  // ── Hydration from reconnect ──
  function hydratePipeline(snapshot) {
    if (!snapshot) return;
    state = reducePipelineEvent(state, snapshot);
    // Replay all stored stage events
    (snapshot.stages || []).forEach(ev => { state = reducePipelineEvent(state, ev); });
    render();
  }
  
  // ── Rendering ──
  function render() { /* render phases, stages, progress rail */ }
  function renderPhase(phaseKey) { /* collapsible phase section */ }
  function renderStageAttempt(stageKey, attempt) { /* stage row with icon, name, retry badge, duration */ }
  
  // ── Init ──
  function init() {
    document.addEventListener('circuitbot:socket-ready', (e) => {
      attachSocket(e.detail.socket);
    });
    // Also check if socket already exists
    if (window.socket) attachSocket(window.socket);
    
    // Keyboard: add to command palette
    document.addEventListener('keydown', (e) => {
      if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
        // Register "View Pipeline Activity" command
      }
    });
  }
  
  // Phase groups
  const PHASES = {
    understand: { label: 'Understand', order: 1, stages: ['clarify','analyze','research','architecture_planner','capability_resolver'] },
    components: { label: 'Components', order: 2, stages: ['select','dependency_expander','deduplicator','constraint_checker','repair','validate','ask_validation_help','post_validate_constraint_checker','freeze_components','datasheet_search'] },
    schematic: { label: 'Schematic', order: 3, stages: ['symbol_compatibility','dispatch','pin_marker','symbol_validate','connection_search','netlist','power_net_repair','structural_net_validate','structural_net_repair','placement','routing','connectivity_validate','connectivity_repair','schematic_audit','schematic_repair'] },
    pcb: { label: 'PCB', order: 4, stages: ['ask_pcb_approval','ask_board_config','pcb_layout','design_review'] },
  };
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
  
  window.pipelinePanel = { state, hydratePipeline, attachSocket };
})();
```
